const fmt = n => new Intl.NumberFormat('vi-VN').format(n)
const getItem = name => screen.getByText(name).closest('article')
const orderRow = id => screen.getByText(`#${id}`).closest('.bc-o')

describe('CustomerTimeline', () => {
  it('hiện thông báo khi không có dữ liệu', () => {
    render(<CustomerTimeline data={[]} />)
    expect(
      screen.getByText('Không có khách nào trong bộ lọc đã chọn.')
    ).toBeInTheDocument()
  })

  it('hiện tiêu đề từng khách: tên, SĐT, giờ check-in, tổng thanh toán', () => {
    render(<CustomerTimeline data={DATA} />)
    const a = getItem('Nguyễn Hùng Huy')
    expect(within(a).getByText('0977629598')).toBeInTheDocument()
    expect(within(a).getByText('--:--')).toBeInTheDocument()
    expect(a.querySelector('.bc-tot').textContent).toBe(fmt(10530000))

    const b = getItem('0000(Vãng lai)')
    expect(within(b).getByText('00:23')).toBeInTheDocument()
    // tổng = 0: ẩn trên desktop bằng class bc-zero
    expect(b.querySelector('.bc-tot')).toHaveClass('bc-zero')
  })

  it('đơn bán mới nhiều phương thức: liệt kê từng phương thức và nợ', () => {
    render(<CustomerTimeline data={DATA} />)
    const row = orderRow(65408)
    expect(within(row).getByLabelText('Bán mới')).toBeInTheDocument()
    expect(within(row).getByText(fmt(14030000))).toBeInTheDocument()
    expect(within(row).getByText('Thực thu')).toBeInTheDocument()
    expect(within(row).getByText(fmt(4030000))).toBeInTheDocument()
    const breakdown = row.querySelector('.bc-v').textContent
    expect(breakdown).toContain(`Tiền mặt: ${fmt(1030000)}`)
    expect(breakdown).toContain(`Chuyển khoản: ${fmt(3000000)}`)
    expect(breakdown).toContain(`Ví: ${fmt(100000)}`)
    expect(breakdown).toContain(`Thẻ tiền: ${fmt(3000000)}`)
    expect(row.querySelector('.bc-nw').textContent).toBe(`Nợ: ${fmt(6900000)}`)
    // tên nhân viên rút gọn, tên đầy đủ trong chú thích
    const nm = within(row).getAllByText('LinhTM')[0]
    expect(nm).toHaveAttribute('data-tip', 'Trần Mai Linh')
  })

  it('đơn một phương thức bằng đúng số tiền: ghi tên phương thức cạnh số tiền', () => {
    render(<CustomerTimeline data={DATA} />)
    const row = orderRow(65407)
    expect(row.querySelector('.bc-mth').textContent).toBe('TM')
    expect(row.querySelector('.bc-sg').textContent).toBe('Tiền mặt')
  })

  it('thanh toán nợ và trả hàng', () => {
    render(<CustomerTimeline data={DATA} />)
    const debt = orderRow(65412)
    expect(within(debt).getByLabelText('Thanh toán nợ')).toBeInTheDocument()
    expect(debt.querySelector('.bc-mth').textContent).toBe('CK')

    const ret = orderRow(65410)
    expect(within(ret).getByLabelText('Trả hàng')).toBeInTheDocument()
    expect(within(ret).getByText('Hoàn tiền')).toHaveClass('bc-ret')
    expect(within(ret).getByText(fmt(4000000))).toHaveClass('bc-neg')
    expect(within(ret).getByLabelText('Khấu trừ hoa hồng')).toBeInTheDocument()
    expect(ret.textContent).toContain(`(-${fmt(976000)})`)
  })

  it('đơn chưa thanh toán: không có số thực thu, chỉ hiện nợ', () => {
    render(<CustomerTimeline data={DATA} />)
    const row = orderRow(65411)
    expect(row.querySelector('.bc-top')).toBeNull()
    expect(row.querySelector('.bc-nw').textContent).toBe(`Nợ: ${fmt(2300000)}`)
  })

  it('cột dịch vụ và nghiệp vụ khác', () => {
    render(<CustomerTimeline data={DATA} />)
    const a = getItem('Nguyễn Hùng Huy')
    expect(within(a).getByText('Lớp tập tô')).toBeInTheDocument()
    expect(
      within(a).getByText('Phụ phí xăng xe - dưới 5km(Gốc)')
    ).toBeInTheDocument()
    expect(within(a).getByText(`+${fmt(3000000)}`)).toHaveClass('bc-pos')
    expect(within(a).getByText(`-${fmt(1000000)}`)).toHaveClass('bc-neg')
    expect(within(a).getByText('Không tính vào doanh thu')).toBeInTheDocument()

    // khách không có dịch vụ / nghiệp vụ khác: cột hiện "Không có" và có class bc-none
    const b = getItem('0000(Vãng lai)')
    expect(within(b).getAllByText('Không có')).toHaveLength(2)
    expect(b.querySelectorAll('.bc-col.bc-none')).toHaveLength(2)
  })

  it('bấm tiêu đề để thu gọn / mở lại', () => {
    render(<CustomerTimeline data={DATA} />)
    const a = getItem('Nguyễn Hùng Huy')
    const head = a.querySelector('.bc-head')
    expect(head).toHaveAttribute('aria-expanded', 'true')
    fireEvent.click(head)
    expect(head).toHaveAttribute('aria-expanded', 'false')
    expect(a.querySelector('.bc-cols')).toBeNull()
    fireEvent.click(head)
    expect(a.querySelector('.bc-cols')).not.toBeNull()
  })

  it('chạm tên nhân viên để hiện chú thích, chạm chỗ khác để ẩn', () => {
    render(<CustomerTimeline data={DATA} />)
    const nm = within(orderRow(65408)).getAllByText('LinhTM')[0]
    fireEvent.click(nm)
    expect(nm).toHaveClass('bc-show')
    fireEvent.click(document.body)
    expect(nm).not.toHaveClass('bc-show')
  })
})
